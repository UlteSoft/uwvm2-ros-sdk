cmake_minimum_required(VERSION 3.20)
project(uwvm_nvidia_cpu_buffer_exchange LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Real Linux ROS 2 native exchange required")
endif()
option(UWVM_NVIDIA_ENABLE_CPU_EXCHANGE "Explicit loopback-only native exchange" OFF)
option(UWVM_NVIDIA_CPU_CUDA "Build genuine CUDA handle/promotion checks" OFF)
foreach(_name UWVM_CPU_SENSOR_INCLUDE UWVM_ROSIDL_BUFFER_INCLUDE)
  if(NOT IS_ABSOLUTE "${${_name}}")
    message(FATAL_ERROR "Explicit installed include root required: ${_name}")
  endif()
endforeach()
if(NOT EXISTS "${UWVM_CPU_SENSOR_INCLUDE}/cpu_sensor_contract.h")
  message(FATAL_ERROR "Actual complete CPU sensor contract is missing")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rosidl_buffer 5.3.1 EXACT REQUIRED)
set(_pins --rosidl-include "${UWVM_ROSIDL_BUFFER_INCLUDE}")
if(UWVM_NVIDIA_CPU_CUDA)
  if(NOT IS_ABSOLUTE "${UWVM_CUDA_BUFFER_INCLUDE}")
    message(FATAL_ERROR "Explicit installed CUDA buffer include root required")
  endif()
  find_package(cuda_buffer REQUIRED)
  find_package(CUDAToolkit 11.8 REQUIRED)
  list(APPEND _pins --cuda-include "${UWVM_CUDA_BUFFER_INCLUDE}")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" -I -S -B
  "${UWVM_SDK_ROOT}/tools/platform/nvidia-isaac-ros/native/verify_api.py" ${_pins}
  --output "${CMAKE_CURRENT_BINARY_DIR}/api-headers.json"
  RESULT_VARIABLE _pin_status)
if(NOT _pin_status EQUAL 0)
  message(FATAL_ERROR "Pinned actual Buffer API read-back failed; no compatibility substitute")
endif()
foreach(_target rclcpp::rclcpp rosidl_buffer::rosidl_buffer
    sensor_msgs::sensor_msgs__rosidl_generator_cpp sensor_msgs::sensor_msgs__rosidl_typesupport_cpp)
  if(NOT TARGET "${_target}")
    message(FATAL_ERROR "Actual installed native target missing: ${_target}")
  endif()
endforeach()
add_executable(uwvm-nvidia-cpu-exchange exchange.cpp)
target_include_directories(uwvm-nvidia-cpu-exchange BEFORE PRIVATE
  "${UWVM_SDK_ROOT}/include" "${UWVM_CPU_SENSOR_INCLUDE}" "${UWVM_ROSIDL_BUFFER_INCLUDE}")
target_link_libraries(uwvm-nvidia-cpu-exchange PRIVATE rclcpp::rclcpp rosidl_buffer::rosidl_buffer
  sensor_msgs::sensor_msgs__rosidl_generator_cpp sensor_msgs::sensor_msgs__rosidl_typesupport_cpp)
target_compile_definitions(uwvm-nvidia-cpu-exchange PRIVATE
  UWVM_NVIDIA_ENABLE_CPU_EXCHANGE=$<BOOL:${UWVM_NVIDIA_ENABLE_CPU_EXCHANGE}>
  UWVM_NVIDIA_CPU_CUDA=$<BOOL:${UWVM_NVIDIA_CPU_CUDA}>)
if(UWVM_NVIDIA_CPU_CUDA)
  if(NOT TARGET cuda_buffer::cuda_buffer)
    message(FATAL_ERROR "Actual exported cuda_buffer::cuda_buffer target required")
  endif()
  target_include_directories(uwvm-nvidia-cpu-exchange BEFORE PRIVATE "${UWVM_CUDA_BUFFER_INCLUDE}")
  target_link_libraries(uwvm-nvidia-cpu-exchange PRIVATE cuda_buffer::cuda_buffer CUDA::cudart)
endif()

# Public language imports are resolved from the installed global SDK roots.
if(DEFINED UWVM_SDK_ROOT)
  include_directories("${UWVM_SDK_ROOT}/include/platform")
endif()
