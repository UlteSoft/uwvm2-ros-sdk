cmake_minimum_required(VERSION 3.20)
project(uwvm_nvidia_isaac_ros_guest LANGUAGES CXX)
if(NOT CMAKE_CXX_STANDARD)
 set(CMAKE_CXX_STANDARD 20)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "WASI")
 message(FATAL_ERROR "Actual selected wasm32 WASI toolchain required")
endif()
set(_cpp_foundation "${UWVM_SDK_ROOT}/include/wasm/language/c++/include")
if(NOT IS_ABSOLUTE "${UWVM_SDK_ROOT}" OR NOT EXISTS "${UWVM_SDK_ROOT}/include/platform/ros2/c/roswasm/ros2.h" OR NOT EXISTS "${_cpp_foundation}/fast_io.h" OR NOT EXISTS "${_cpp_foundation}/c++/v1/__config_site")
 message(FATAL_ERROR "Explicit installed SDK with the actual I/O/header closure required")
endif()
if(NOT IS_ABSOLUTE "${UWVM_NVIDIA_BOUND_BUNDLE}" OR NOT EXISTS "${UWVM_NVIDIA_BOUND_BUNDLE}/bounded_records.hpp")
 message(FATAL_ERROR "Actual installed complete generated record family required")
endif()
option(UWVM_NVIDIA_POSITIVE "Explicit distinct positive artifact; never a Runtime grant" OFF)
add_executable(nvidia-isaac_ros-telemetry telemetry.cpp)
target_include_directories(nvidia-isaac_ros-telemetry PRIVATE "${UWVM_SDK_ROOT}/include" "${_cpp_foundation}" "${UWVM_SDK_ROOT}/include/platform/ros2/c" "${UWVM_SDK_ROOT}/include/platform/ros/cpp" "${UWVM_NVIDIA_BOUND_BUNDLE}")
target_compile_definitions(nvidia-isaac_ros-telemetry PRIVATE UWVM_NVIDIA_POSITIVE=$<BOOL:${UWVM_NVIDIA_POSITIVE}>)
add_executable(nvidia-raw-denied raw_denied.cpp)
target_include_directories(nvidia-raw-denied PRIVATE "${UWVM_SDK_ROOT}/include" "${_cpp_foundation}" "${UWVM_SDK_ROOT}/include/platform/ros2/c")
set_target_properties(nvidia-isaac_ros-telemetry nvidia-raw-denied PROPERTIES SUFFIX ".wasm")

# Public language imports are resolved from the installed global SDK roots.
if(DEFINED UWVM_SDK_ROOT)
  include_directories("${UWVM_SDK_ROOT}/include/platform")
endif()
