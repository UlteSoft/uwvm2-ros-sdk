cmake_minimum_required(VERSION 3.20)
project(uwvm_sdk_native_test_peer LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
add_executable(uwvm-sdk-ros2-component-peer ros2_component_peer.cpp)
target_compile_features(uwvm-sdk-ros2-component-peer PRIVATE cxx_std_17)
target_compile_options(uwvm-sdk-ros2-component-peer PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(uwvm-sdk-ros2-component-peer rclcpp std_msgs std_srvs)
