cmake_minimum_required(VERSION 3.22)
project(uwvm_official_physics_io LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
if(NOT IS_DIRECTORY "${MUJOCO_INCLUDE_DIR}" OR NOT EXISTS "${MUJOCO_LIBRARY}")
  message(FATAL_ERROR "Select existing official MuJoCo headers/library; no downloads are performed")
endif()
add_executable(uwvm-mujoco-io mujoco_io.cc)
target_compile_features(uwvm-mujoco-io PRIVATE cxx_std_20)
target_include_directories(uwvm-mujoco-io PRIVATE "${MUJOCO_INCLUDE_DIR}")
target_link_libraries(uwvm-mujoco-io "${MUJOCO_LIBRARY}")
ament_target_dependencies(uwvm-mujoco-io rclcpp sensor_msgs)
target_compile_options(uwvm-mujoco-io PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS uwvm-mujoco-io DESTINATION bin)

if(WEBOTS_HOME)
  if(NOT EXISTS "${WEBOTS_HOME}/include/controller/c/webots/robot.h" OR
     NOT EXISTS "${WEBOTS_HOME}/lib/controller/libController.so")
    message(FATAL_ERROR "Select an existing official Webots installation")
  endif()
  add_executable(uwvm-webots-io webots_io.cc)
  target_compile_features(uwvm-webots-io PRIVATE cxx_std_20)
  target_include_directories(uwvm-webots-io PRIVATE "${WEBOTS_HOME}/include/controller/c")
  target_link_libraries(uwvm-webots-io "${WEBOTS_HOME}/lib/controller/libController.so")
  ament_target_dependencies(uwvm-webots-io rclcpp sensor_msgs)
  target_compile_options(uwvm-webots-io PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS uwvm-webots-io DESTINATION bin)
endif()
