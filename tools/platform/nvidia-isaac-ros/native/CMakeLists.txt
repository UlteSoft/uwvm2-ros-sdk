cmake_minimum_required(VERSION 3.20)
project(uwvm_nvidia_isaac_ros_export LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT IS_ABSOLUTE "${UWVM_SDK_ROOT}" OR NOT EXISTS "${UWVM_SDK_ROOT}/tools/custom-types/native_current/SelectedNative.cmake")
 message(FATAL_ERROR "Explicit installed SDK with the canonical native CMake helper required")
endif()
include("${UWVM_SDK_ROOT}/tools/custom-types/native_current/SelectedNative.cmake")
if(NOT IS_ABSOLUTE "${UWVM_NVIDIA_BUNDLE}")
 message(FATAL_ERROR "Explicit actual generated whole bundle required")
endif()
foreach(_leaf native_identity.cc native_constants.hpp generation.vendor.json)
 if(NOT EXISTS "${UWVM_NVIDIA_BUNDLE}/${_leaf}")
  message(FATAL_ERROR "Actual vendor product missing: ${_leaf}")
 endif()
endforeach()
find_package(ament_cmake REQUIRED)
find_package(rosidl_runtime_c REQUIRED)
find_package(rosidl_typesupport_cpp REQUIRED)
set(_targets rosidl_runtime_c::rosidl_runtime_c rosidl_typesupport_cpp::rosidl_typesupport_cpp)
find_package(isaac_ros_apriltag_interfaces REQUIRED)
if(NOT TARGET isaac_ros_apriltag_interfaces::isaac_ros_apriltag_interfaces__rosidl_generator_cpp)
 message(FATAL_ERROR "Real installed target missing: isaac_ros_apriltag_interfaces::isaac_ros_apriltag_interfaces__rosidl_generator_cpp")
endif()
list(APPEND _targets isaac_ros_apriltag_interfaces::isaac_ros_apriltag_interfaces__rosidl_generator_cpp)
if(NOT TARGET isaac_ros_apriltag_interfaces::isaac_ros_apriltag_interfaces__rosidl_typesupport_cpp)
 message(FATAL_ERROR "Real installed target missing: isaac_ros_apriltag_interfaces::isaac_ros_apriltag_interfaces__rosidl_typesupport_cpp")
endif()
list(APPEND _targets isaac_ros_apriltag_interfaces::isaac_ros_apriltag_interfaces__rosidl_typesupport_cpp)
find_package(geometry_msgs REQUIRED)
if(NOT TARGET geometry_msgs::geometry_msgs__rosidl_generator_cpp)
 message(FATAL_ERROR "Real installed target missing: geometry_msgs::geometry_msgs__rosidl_generator_cpp")
endif()
list(APPEND _targets geometry_msgs::geometry_msgs__rosidl_generator_cpp)
if(NOT TARGET geometry_msgs::geometry_msgs__rosidl_typesupport_cpp)
 message(FATAL_ERROR "Real installed target missing: geometry_msgs::geometry_msgs__rosidl_typesupport_cpp")
endif()
list(APPEND _targets geometry_msgs::geometry_msgs__rosidl_typesupport_cpp)
find_package(std_msgs REQUIRED)
if(NOT TARGET std_msgs::std_msgs__rosidl_generator_cpp)
 message(FATAL_ERROR "Real installed target missing: std_msgs::std_msgs__rosidl_generator_cpp")
endif()
list(APPEND _targets std_msgs::std_msgs__rosidl_generator_cpp)
if(NOT TARGET std_msgs::std_msgs__rosidl_typesupport_cpp)
 message(FATAL_ERROR "Real installed target missing: std_msgs::std_msgs__rosidl_typesupport_cpp")
endif()
list(APPEND _targets std_msgs::std_msgs__rosidl_typesupport_cpp)
find_package(builtin_interfaces REQUIRED)
if(NOT TARGET builtin_interfaces::builtin_interfaces__rosidl_generator_cpp)
 message(FATAL_ERROR "Real installed target missing: builtin_interfaces::builtin_interfaces__rosidl_generator_cpp")
endif()
list(APPEND _targets builtin_interfaces::builtin_interfaces__rosidl_generator_cpp)
if(NOT TARGET builtin_interfaces::builtin_interfaces__rosidl_typesupport_cpp)
 message(FATAL_ERROR "Real installed target missing: builtin_interfaces::builtin_interfaces__rosidl_typesupport_cpp")
endif()
list(APPEND _targets builtin_interfaces::builtin_interfaces__rosidl_typesupport_cpp)
uwvm_current_native_link_environment(${_targets})
add_executable(uwvm-nvidia-isaac_ros-export export_identities.cpp)
target_include_directories(uwvm-nvidia-isaac_ros-export PRIVATE "${UWVM_SDK_ROOT}" "${UWVM_NVIDIA_BUNDLE}")
target_compile_definitions(uwvm-nvidia-isaac_ros-export PRIVATE UWVM_CUSTOM_NATIVE_ROS_VERSION=2)
target_link_libraries(uwvm-nvidia-isaac_ros-export PRIVATE ${_targets})

# Native application targets were removed from the public SDK pending real
# all-Wasm migration. Supplying an old enable must fail before any native build.
if(UWVM_NVIDIA_BUILD_DETECTOR OR UWVM_NVIDIA_ENABLE_NATIVE_DETECTOR)
 message(FATAL_ERROR "BLOCKED_LEGACY_ROS_TRANSPLATE: native detector application removed; use qualified guest migration")
